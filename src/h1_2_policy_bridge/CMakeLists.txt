cmake_minimum_required(VERSION 3.16)
project(h1_2_policy_bridge)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(h1_2_policy_interfaces REQUIRED)
find_package(Eigen3 REQUIRED)
if(CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
 set(ORT_ARCH x64)
elseif(CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
 set(ORT_ARCH aarch64)
else()
 message(FATAL_ERROR "Unsupported architecture ${CMAKE_SYSTEM_PROCESSOR}")
endif()
set(ORT_ROOT "/home/llx/unitree_rl_mjlab/deploy/thirdparty/onnxruntime-linux-${ORT_ARCH}-1.22.0" CACHE PATH "ONNX Runtime directory")
set(UNITREE_SDK2_ROOT "/home/llx/unitree_sdk2" CACHE PATH "Unitree SDK2")
add_library(h1_policy_core SHARED src/core.cpp src/ort.cpp src/controller.cpp src/trained_guard.cpp)
target_include_directories(h1_policy_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE ${ORT_ROOT}/include)
target_link_libraries(h1_policy_core ${ORT_ROOT}/lib/libonnxruntime.so.1.22.0)
set_target_properties(h1_policy_core PROPERTIES BUILD_RPATH "${ORT_ROOT}/lib" INSTALL_RPATH "${ORT_ROOT}/lib")
find_package(Threads REQUIRED)
add_library(h1_policy_node_impl SHARED src/node.cpp src/io.cpp)
target_include_directories(h1_policy_node_impl PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> ${UNITREE_SDK2_ROOT}/include ${UNITREE_SDK2_ROOT}/thirdparty/include ${UNITREE_SDK2_ROOT}/thirdparty/include/ddscxx)
ament_target_dependencies(h1_policy_node_impl rclcpp rcl_interfaces geometry_msgs std_srvs std_msgs h1_2_policy_interfaces)
target_link_libraries(h1_policy_node_impl Eigen3::Eigen h1_policy_core ${UNITREE_SDK2_ROOT}/lib/${CMAKE_SYSTEM_PROCESSOR}/libunitree_sdk2.a ${UNITREE_SDK2_ROOT}/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}/libddscxx.so ${UNITREE_SDK2_ROOT}/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}/libddsc.so Threads::Threads)
target_link_options(h1_policy_node_impl PRIVATE -Wl,--disable-new-dtags)
set_target_properties(h1_policy_node_impl PROPERTIES BUILD_RPATH "${UNITREE_SDK2_ROOT}/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}" INSTALL_RPATH "${UNITREE_SDK2_ROOT}/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}")
add_executable(h1_2_policy_node src/main.cpp)
ament_target_dependencies(h1_2_policy_node rclcpp geometry_msgs std_srvs std_msgs h1_2_policy_interfaces)
target_link_options(h1_2_policy_node PRIVATE -Wl,--disable-new-dtags)
target_link_libraries(h1_2_policy_node h1_policy_node_impl)
set_target_properties(h1_2_policy_node PROPERTIES BUILD_RPATH "${UNITREE_SDK2_ROOT}/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}" INSTALL_RPATH "${UNITREE_SDK2_ROOT}/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}")
install(TARGETS h1_2_policy_node h1_policy_node_impl RUNTIME DESTINATION lib/${PROJECT_NAME} LIBRARY DESTINATION lib)
add_executable(h1_policy_regression test/onnx_regression.cpp)
target_link_libraries(h1_policy_regression h1_policy_core)
if(BUILD_TESTING)
 add_executable(h1_policy_dds_smoke test/dds_smoke.cpp)
 target_include_directories(h1_policy_dds_smoke PRIVATE ${UNITREE_SDK2_ROOT}/include ${UNITREE_SDK2_ROOT}/thirdparty/include ${UNITREE_SDK2_ROOT}/thirdparty/include/ddscxx)
 target_link_options(h1_policy_dds_smoke PRIVATE -Wl,--disable-new-dtags)
 target_link_libraries(h1_policy_dds_smoke ${UNITREE_SDK2_ROOT}/lib/${CMAKE_SYSTEM_PROCESSOR}/libunitree_sdk2.a ${UNITREE_SDK2_ROOT}/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}/libddscxx.so ${UNITREE_SDK2_ROOT}/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}/libddsc.so Threads::Threads)
 set_target_properties(h1_policy_dds_smoke PROPERTIES BUILD_RPATH "${UNITREE_SDK2_ROOT}/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}")
 enable_testing()
 set(H1_POLICY_TEST_MODEL_PATH "/home/llx/policy.onnx" CACHE FILEPATH "Model used by tests")
 add_executable(h1_policy_node_integration_test test/node_integration_test.cpp)
 ament_target_dependencies(h1_policy_node_integration_test rclcpp geometry_msgs std_srvs std_msgs h1_2_policy_interfaces)
 target_link_options(h1_policy_node_integration_test PRIVATE -Wl,--disable-new-dtags)
 target_link_libraries(h1_policy_node_integration_test h1_policy_node_impl)
 set_target_properties(h1_policy_node_integration_test PROPERTIES BUILD_RPATH "${UNITREE_SDK2_ROOT}/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}")
 add_test(NAME node_integration COMMAND h1_policy_node_integration_test ${H1_POLICY_TEST_MODEL_PATH})
 set_tests_properties(node_integration PROPERTIES ENVIRONMENT "ROS_LOG_DIR=${CMAKE_CURRENT_BINARY_DIR}/ros-log;ROS_LOCALHOST_ONLY=1;ROS_DOMAIN_ID=73;RMW_IMPLEMENTATION=rmw_fastrtps_cpp")
 add_executable(h1_policy_hardware_test test/hardware_test.cpp)
 target_include_directories(h1_policy_hardware_test PRIVATE include ${UNITREE_SDK2_ROOT}/include)
 add_test(NAME hardware COMMAND h1_policy_hardware_test)
 add_test(NAME launch_parameters COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/test/test_policy_launch.py)
 set_tests_properties(launch_parameters PROPERTIES ENVIRONMENT "ROS_LOG_DIR=/tmp")
 add_executable(h1_policy_trained_guard_test test/trained_guard_test.cpp)
 target_link_libraries(h1_policy_trained_guard_test h1_policy_core)
 target_compile_options(h1_policy_trained_guard_test PRIVATE -UNDEBUG)
 add_test(NAME trained_guard COMMAND h1_policy_trained_guard_test)
 add_executable(h1_policy_core_test test/core_test.cpp)
 target_link_libraries(h1_policy_core_test h1_policy_core)
 add_test(NAME core COMMAND h1_policy_core_test ${H1_POLICY_TEST_MODEL_PATH})
 add_test(NAME onnx_regression COMMAND h1_policy_regression ${H1_POLICY_TEST_MODEL_PATH})
endif()
install(TARGETS h1_policy_core h1_policy_regression EXPORT export_h1_policy_core LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_h1_policy_core HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_package()
